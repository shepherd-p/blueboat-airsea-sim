cmake_minimum_required(VERSION 3.16)
project(blueboat_video_capture)
find_package(gz_sim_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-sim8-gui REQUIRED)
find_package(gz_gui_vendor REQUIRED)
find_package(gz-gui8 REQUIRED)
find_package(gz_rendering_vendor REQUIRED)
find_package(gz-rendering8 REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
find_package(Qt5 REQUIRED COMPONENTS Core Quick)
set(CMAKE_AUTORCC ON)
add_library(BlueBoatVideoCapture SHARED capture.cc capture.qrc)
target_compile_features(BlueBoatVideoCapture PUBLIC cxx_std_17)
target_link_libraries(BlueBoatVideoCapture gz-sim8::gz-sim8-gui gz-gui8::gz-gui8 gz-rendering8::gz-rendering8 gz-plugin2::register Qt5::Core Qt5::Quick)
add_library(BlueBoatMarkerInspection SHARED marker_inspection.cc marker_inspection.qrc)
target_compile_features(BlueBoatMarkerInspection PUBLIC cxx_std_17)
target_link_libraries(BlueBoatMarkerInspection gz-sim8::gz-sim8-gui gz-gui8::gz-gui8 gz-rendering8::gz-rendering8 gz-plugin2::register Qt5::Core Qt5::Quick)
add_executable(VideoPersistenceCheck EXCLUDE_FROM_ALL persistence_check.cc)
target_compile_features(VideoPersistenceCheck PUBLIC cxx_std_17)
target_link_libraries(VideoPersistenceCheck gz-sim8::gz-sim8-gui)

find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs calib3d aruco)
add_library(BlueBoatTagPose SHARED tag_pose.cc tag_pose.qrc)
set_target_properties(BlueBoatTagPose PROPERTIES AUTOMOC ON)
target_compile_features(BlueBoatTagPose PUBLIC cxx_std_17)
target_include_directories(BlueBoatTagPose PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(BlueBoatTagPose gz-sim8::gz-sim8-gui gz-gui8::gz-gui8 gz-rendering8::gz-rendering8 gz-plugin2::register Qt5::Core Qt5::Quick ${OpenCV_LIBS})
