cmake_minimum_required(VERSION 3.16)
project(blueboat_gazebo)
find_package(ament_cmake REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
find_package(gz_transport_vendor REQUIRED)
find_package(gz-transport13 REQUIRED)
find_package(gz-sim8-gui REQUIRED)
find_package(gz_gui_vendor REQUIRED)
find_package(gz-gui8 REQUIRED)
find_package(gz_rendering_vendor REQUIRED)
find_package(gz-rendering8 REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Core Quick)
set(CMAKE_AUTORCC ON)
add_library(BlueBoatVisuals SHARED src/BlueBoatVisuals.cc src/visuals.qrc)
target_compile_features(BlueBoatVisuals PUBLIC cxx_std_17)
target_link_libraries(BlueBoatVisuals gz-sim8::gz-sim8-gui gz-gui8::gz-gui8 gz-rendering8::gz-rendering8 gz-plugin2::register Qt5::Core Qt5::Quick)
install(TARGETS BlueBoatVisuals LIBRARY DESTINATION lib)
add_library(blueboat-buoyancy-system SHARED src/BlueBoatBuoyancy.cc)
target_compile_features(blueboat-buoyancy-system PUBLIC cxx_std_17)
target_link_libraries(blueboat-buoyancy-system gz-sim8::gz-sim8 gz-plugin2::register)
install(TARGETS blueboat-buoyancy-system LIBRARY DESTINATION lib)
add_library(blueboat-thrusters-system SHARED src/BlueBoatThrusters.cc)
target_compile_features(blueboat-thrusters-system PUBLIC cxx_std_17)
target_link_libraries(blueboat-thrusters-system gz-sim8::gz-sim8 gz-plugin2::register gz-transport13::core)
install(TARGETS blueboat-thrusters-system LIBRARY DESTINATION lib)
add_library(blueboat-environment-system SHARED src/BlueBoatEnvironment.cc)
target_compile_features(blueboat-environment-system PUBLIC cxx_std_17)
target_link_libraries(blueboat-environment-system gz-sim8::gz-sim8 gz-plugin2::register gz-transport13::core)
install(TARGETS blueboat-environment-system LIBRARY DESTINATION lib)
add_library(blueboat-sitl-system SHARED src/BlueBoatSITL.cc)
target_compile_features(blueboat-sitl-system PUBLIC cxx_std_17)
target_link_libraries(blueboat-sitl-system gz-sim8::gz-sim8 gz-plugin2::register gz-transport13::core)
install(TARGETS blueboat-sitl-system LIBRARY DESTINATION lib)
ament_package()
