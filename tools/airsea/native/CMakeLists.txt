cmake_minimum_required(VERSION 3.16)
project(airsea_native)
find_package(gz_sim_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-sim8-gui REQUIRED)
find_package(gz_gui_vendor REQUIRED)
find_package(gz-gui8 REQUIRED)
find_package(gz_rendering_vendor REQUIRED)
find_package(gz-rendering8 REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
find_package(Qt5 REQUIRED COMPONENTS Core Quick)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs calib3d aruco)
set(CMAKE_AUTORCC ON)
add_library(AirSeaAircraft SHARED aircraft.cc)
target_compile_features(AirSeaAircraft PUBLIC cxx_std_17)
target_link_libraries(AirSeaAircraft gz-sim8::gz-sim8 gz-plugin2::register)
add_library(AirSeaCapture SHARED capture.cc capture.qrc)
set_target_properties(AirSeaCapture PROPERTIES AUTOMOC ON)
target_compile_features(AirSeaCapture PUBLIC cxx_std_17)
target_include_directories(AirSeaCapture PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(AirSeaCapture gz-sim8::gz-sim8-gui gz-gui8::gz-gui8 gz-rendering8::gz-rendering8 gz-plugin2::register Qt5::Core Qt5::Quick ${OpenCV_LIBS})
